cmake_minimum_required(VERSION 3.24)
project(raisim_vs_newton LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(BENCHMARK_SDK_SOURCE_DIR "" CACHE PATH "Existing public raisim2Lib checkout; empty downloads it")
set(BENCHMARK_SDK_REVISION "d9df2c979c9ddc44ec6b6727ff72c660688fdb62" CACHE STRING "Public SDK commit")
include(cmake/PublicSdk.cmake)
find_package(Threads REQUIRED)
list(PREPEND CMAKE_PREFIX_PATH "${BENCHMARK_SDK_SOURCE_DIR}/thirdParty/Eigen3" "${BENCHMARK_SDK_PREFIX}")
set(_benchmark_expected_config "${BENCHMARK_SDK_PREFIX}/lib/cmake/raisim")
if(raisim_DIR AND NOT raisim_DIR STREQUAL _benchmark_expected_config)
  message(FATAL_ERROR "SDK package path changed; configure a clean build directory.")
endif()
find_package(raisim CONFIG REQUIRED PATHS "${BENCHMARK_SDK_PREFIX}/lib/cmake/raisim" NO_DEFAULT_PATH)
option(BENCHMARK_X86_64_V3 "Match the reference x86-64-v3 driver flags on supported x86 Linux CPUs" OFF)
add_executable(raisim_benchmark src/raisim_benchmark.cpp)
add_executable(manipulation_benchmark src/manipulation_benchmark.cpp)
add_executable(export_model src/export_model.cpp)
foreach(target raisim_benchmark export_model manipulation_benchmark)
  target_link_libraries(${target} PRIVATE raisim::raisim Threads::Threads)
  target_compile_definitions(${target} PRIVATE EIGEN_DONT_PARALLELIZE EIGEN_NO_DEBUG)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -O3 -ffp-contract=off)
  endif()
  if(BENCHMARK_X86_64_V3)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$" OR MSVC)
      message(FATAL_ERROR "BENCHMARK_X86_64_V3 needs an x86-64 GCC/Clang toolchain")
    endif()
    target_compile_options(${target} PRIVATE -march=x86-64-v3 -mtune=generic)
    target_compile_definitions(${target} PRIVATE RAISIM_X86_64_V3)
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(WIN32)
    target_link_libraries(${target} PRIVATE ws2_32 winmm)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:${target}> $<TARGET_FILE_DIR:${target}>
      COMMAND_EXPAND_LISTS)
    file(GLOB sdk_runtime_dlls "${BENCHMARK_SDK_PREFIX}/bin/*.dll" "${BENCHMARK_SDK_PREFIX}/lib/*.dll" "${BENCHMARK_SDK_PREFIX}/bin/openusd/*.dll" "${BENCHMARK_SDK_PREFIX}/lib/openusd/*.dll")
    foreach(dll IN LISTS sdk_runtime_dlls)
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" $<TARGET_FILE_DIR:${target}>)
    endforeach()
    foreach(subdir bin/openusd lib/openusd)
      if(EXISTS "${BENCHMARK_SDK_PREFIX}/${subdir}")
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${BENCHMARK_SDK_PREFIX}/${subdir}" "$<TARGET_FILE_DIR:${target}>")
      endif()
    endforeach()
  endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/benchmark_paths_$<CONFIG>.json" CONTENT
  "{\"sdk\":\"${BENCHMARK_SDK_SOURCE_DIR}\",\"sdk_prefix\":\"${BENCHMARK_SDK_PREFIX}\",\"sdk_revision\":\"${BENCHMARK_SDK_ACTUAL_REVISION}\",\"requested_sdk_revision\":\"${BENCHMARK_SDK_REVISION}\",\"sdk_dirty\":${BENCHMARK_SDK_DIRTY},\"cpu\":\"$<TARGET_FILE:raisim_benchmark>\",\"exporter\":\"$<TARGET_FILE:export_model>\"}\n")
include(CTest)
if(BUILD_TESTING)
  add_executable(price_probe tests/price_probe.cpp)
  target_include_directories(price_probe PRIVATE src)
  add_test(NAME benchmark_contracts COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -v)
  if(WIN32)
    set_tests_properties(benchmark_contracts PROPERTIES ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:${BENCHMARK_SDK_PREFIX}/bin;PATH=path_list_prepend:${BENCHMARK_SDK_PREFIX}/lib;PATH=path_list_prepend:${BENCHMARK_SDK_PREFIX}/bin/openusd;PATH=path_list_prepend:${BENCHMARK_SDK_PREFIX}/lib/openusd")
  endif()
  set_tests_properties(benchmark_contracts PROPERTIES ENVIRONMENT "BENCHMARK_CPU=$<TARGET_FILE:raisim_benchmark>;PRICE_PROBE=$<TARGET_FILE:price_probe>;MANIPULATION_DRIVER=$<TARGET_FILE:manipulation_benchmark>;BENCHMARK_ASSETS=${CMAKE_BINARY_DIR}/assets")
endif()
